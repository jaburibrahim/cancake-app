'' : v; }); if (rowIndex > -1) { sheet.getRange(rowIndex, 1, 1, headers.length).setValues([rowData]); } else { sheet.appendRow(rowData); } } finally { lock.releaseLock(); } return readAllOrders(); } function deleteOrderById(id) { const lock = LockService.getScriptLock();
